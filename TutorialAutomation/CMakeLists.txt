cmake_minimum_required(VERSION 3.20)
project(TutorialAutomation VERSION 1.0 LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "TutorialAutomation requires Windows")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(tutorial_core src/script_engine.c src/window_manager.c src/playback.c src/playback_controls.c src/keyboard.c src/mouse.c src/coding_action.c src/typing_pace.c src/browser_action.c src/browser_scroll_plan.c src/action_pace.c src/ui_targets.c src/presentation.c src/json_config.c src/reference_config.c src/profile.c src/human_motion.c src/web_policy.c src/browse_catalog.c src/web_page.c src/browser_session.c src/project_reference.c src/project_manifest.c src/project_action.c src/side_apps.c src/loggy3.c src/work_session.c src/flow_config.c src/coding_schedule.c src/saved_text_wait.c src/vscode_editor.c)
target_include_directories(tutorial_core PUBLIC include)
target_compile_definitions(tutorial_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(tutorial_core PUBLIC user32 wtsapi32 shell32 ole32 oleaut32 oleacc bcrypt)
if(MSVC)
  target_compile_options(tutorial_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(tutorial_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(tutorial src/main.c)
target_link_libraries(tutorial PRIVATE tutorial_core shell32)
add_executable(monitor src/monitor.c)
